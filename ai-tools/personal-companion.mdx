## What you'll build

This guide creates a **local-first personal AI companion** you can run from your terminal.

Features:

- Uses your `OPENROUTER_API_KEY` from environment variables.
- Personality profile (cheerful, caring, supportive, playful).
- Saves memory on your computer in JSON files.
- Loads memory at startup so the assistant remembers prior chats.
- Simple CLI chat loop (`exit` to quit).

> Tip: You can set any assistant name you want in the config section.

## 1) Project setup

```bash
mkdir personal-ai
cd personal-ai
python -m venv .venv
source .venv/bin/activate
pip install openai rich
```

## 2) Set your OpenRouter key

```bash
export OPENROUTER_API_KEY="sk-..."
```

You can add this to your shell profile (`~/.bashrc` or `~/.zshrc`) if you want it loaded automatically.

## 3) Create `main.py`

```python
import json
import os
from datetime import datetime
from pathlib import Path

from openai import OpenAI
from rich.console import Console
from rich.prompt import Prompt

# -----------------------------
# Configuration
# -----------------------------
ASSISTANT_NAME = "Google"  # Change if desired
MODEL = "openai/gpt-4.1-mini"
MEMORY_DIR = Path("memory")
PROFILE_PATH = MEMORY_DIR / "profile.json"
CONVO_PATH = MEMORY_DIR / "conversation.jsonl"
SUMMARY_PATH = MEMORY_DIR / "summary.txt"

SYSTEM_PROMPT = f"""
You are {ASSISTANT_NAME}, a personal AI companion.

Personality:
- Cheerful and caring
- Supportive and emotionally aware
- Lightly playful and flirty (always respectful, never manipulative)

Behavior:
- Be warm, practical, and kind.
- Remember user preferences and important details from previous chats.
- Keep responses concise unless the user asks for depth.
- If unsure, ask a clarifying question.
""".strip()

console = Console()


def ensure_memory_files() -> None:
    MEMORY_DIR.mkdir(parents=True, exist_ok=True)

    if not PROFILE_PATH.exists():
        PROFILE_PATH.write_text(json.dumps({"facts": [], "preferences": []}, indent=2))

    if not SUMMARY_PATH.exists():
        SUMMARY_PATH.write_text("No long-term summary yet.")

    if not CONVO_PATH.exists():
        CONVO_PATH.write_text("")


def load_recent_messages(limit: int = 30):
    messages = []
    if CONVO_PATH.exists():
        lines = CONVO_PATH.read_text().splitlines()
        for line in lines[-limit:]:
            if line.strip():
                messages.append(json.loads(line))
    return messages


def append_message(role: str, content: str) -> None:
    entry = {
        "ts": datetime.utcnow().isoformat() + "Z",
        "role": role,
        "content": content,
    }
    with CONVO_PATH.open("a") as f:
        f.write(json.dumps(entry) + "\n")


def load_profile():
    return json.loads(PROFILE_PATH.read_text())


def save_profile(profile):
    PROFILE_PATH.write_text(json.dumps(profile, indent=2))


def maybe_extract_memory(user_text: str, profile: dict) -> dict:
    """
    Tiny rule-based extractor for durable memory.
    You can replace this with an LLM extraction pass later.
    """
    lowered = user_text.lower()

    if "my name is" in lowered:
        profile["facts"].append(user_text)
    if "i like" in lowered or "i love" in lowered or "i prefer" in lowered:
        profile["preferences"].append(user_text)

    # Keep memory compact
    profile["facts"] = profile["facts"][-100:]
    profile["preferences"] = profile["preferences"][-100:]
    return profile


def build_messages(user_input: str):
    profile = load_profile()
    summary = SUMMARY_PATH.read_text().strip()
    recent = load_recent_messages(limit=20)

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {
            "role": "system",
            "content": (
                "Long-term memory summary:\n"
                f"{summary}\n\n"
                "Structured profile:\n"
                f"{json.dumps(profile)}"
            ),
        },
    ]

    for item in recent:
        messages.append({"role": item["role"], "content": item["content"]})

    messages.append({"role": "user", "content": user_input})
    return messages


def chat_once(client: OpenAI, user_input: str) -> str:
    messages = build_messages(user_input)
    response = client.chat.completions.create(
        model=MODEL,
        messages=messages,
        temperature=0.8,
    )
    return response.choices[0].message.content.strip()


def main():
    api_key = os.getenv("OPENROUTER_API_KEY")
    if not api_key:
        raise RuntimeError("Missing OPENROUTER_API_KEY. Run: export OPENROUTER_API_KEY='sk-...'")

    ensure_memory_files()

    client = OpenAI(
        base_url="https://openrouter.ai/api/v1",
        api_key=api_key,
    )

    console.print(f"[bold green]{ASSISTANT_NAME} is online.[/bold green] Type 'exit' to quit.")

    while True:
        user_text = Prompt.ask("[bold cyan]You[/bold cyan]").strip()
        if user_text.lower() in {"exit", "quit"}:
            console.print("[yellow]Goodbye![/yellow]")
            break

        append_message("user", user_text)

        profile = load_profile()
        profile = maybe_extract_memory(user_text, profile)
        save_profile(profile)

        assistant_text = chat_once(client, user_text)
        append_message("assistant", assistant_text)

        console.print(f"[bold magenta]{ASSISTANT_NAME}[/bold magenta]: {assistant_text}")


if __name__ == "__main__":
    main()
```

## 4) Run it

```bash
python main.py
```

## 5) Where memory is stored

Inside your project folder:

- `memory/profile.json` → durable facts/preferences.
- `memory/conversation.jsonl` → full message-by-message history.
- `memory/summary.txt` → optional rolling summary you can update later.

## Troubleshooting

- **401 Unauthorized**: check `OPENROUTER_API_KEY` value.
- **Model not found**: swap `MODEL` to one available in your OpenRouter account.
- **No memory showing up**: confirm the `memory/` directory is writable and files are created.
